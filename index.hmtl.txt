  // ==========================================
  // ANTI-RESTART, TAB LOCK & STATE PERSISTENCE
  // ==========================================
  const STORAGE_KEY = 'SAR_A0_FINAL_EXAM_UNIFIED_V1';
  let examState = {
    started: false,
    submitted: false,
    remainingSeconds: 120 * 60,
    answers: {},
    autoDetectedWords: 0,
    autoDetectedSeconds: 0,
    tabSwitches: 0
  };

  function loadState() {
    const saved = localStorage.getItem(STORAGE_KEY);
    if (saved) {
      try { examState = JSON.parse(saved); } catch (e) {}
    }
  }

  function saveState() {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(examState));
  }

  function lockExamOnFirstInteraction() {
    if (!examState.started) {
      examState.started = true;
      document.getElementById('lockBadge').style.display = 'inline-block';
      saveState();
    }
  }

  // Detect if mobile user switches to translator/dictionary apps
  document.addEventListener('visibilitychange', () => {
    if (document.hidden && examState.started && !examState.submitted) {
      examState.tabSwitches++;
      saveState();
    }
  });

  // Prevent browser reload/exit once started
  window.addEventListener('beforeunload', function (e) {
    if (examState.started && !examState.submitted) {
      e.preventDefault();
      e.returnValue = 'Exam is in progress. You cannot restart once you have started answering.';
    }
  });

  window.addEventListener('keydown', function (e) {
    if (examState.started && !examState.submitted) {
      if (e.key === 'F5' || ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'r')) {
        e.preventDefault();
        alert('Restarting is disabled once you have begun answering the exam.');
      }
    }
  });

  // VOICE RECORDING & SPEECH RECOGNITION
  // ==========================================
  const referencePassageWords = new Set(
    "maya lives in a small flat with her family she often wakes up at 6 she drinks a glass of water and makes hot coffee then she gets ready for work maya works as a cashier at a local shop she starts work at 8 am at work she scans items and handles cash and bank cards she likes her job her co workers are very friendly at 12 pm maya eats lunch with her friend she often brings a sandwich and an apple from home maya finishes work at 4 pm after work she walks to the market to get some bread and milk at home she cooks dinner for her family they eat at 7 pm and talk about their day before bed maya reads a book she goes to sleep at ten o clock she is happy with her daily routine".split(" ")
  );

  let mediaRecorder;
  let audioChunks = [];
  let isRecording = false;
  let speechRecognizer = null;
  let recordingStartTime = 0;

  if ('webkitSpeechRecognition' in window || 'SpeechRecognition' in window) {
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    speechRecognizer = new SpeechRecognition();
    speechRecognizer.continuous = true;
    speechRecognizer.interimResults = true;
    speechRecognizer.lang = 'en-US';

    speechRecognizer.onresult = (event) => {
      let transcript = '';
      for (let i = 0; i < event.results.length; ++i) {
        transcript += ' ' + event.results[i][0].transcript;
      }
      const spokenWords = transcript.toLowerCase().replace(/[^a-z0-9\s]/g, '').trim().split(/\s+/).filter(Boolean);
      // Count matched words up to 145
      let matched = 0;
      spokenWords.forEach(w => {
        if (referencePassageWords.has(w) || w.length >= 2) matched++;
      });
      examState.autoDetectedWords = Math.min(145, matched);
      saveState();
    };
  }

  const recordBtn = document.getElementById('recordBtn');
  const recordStatus = document.getElementById('recordStatus');
  const audioPlayback = document.getElementById('audioPlayback');

  recordBtn.addEventListener('click', async () => {
    lockExamOnFirstInteraction();
    if (!isRecording) {
      try {
        const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
        mediaRecorder = new MediaRecorder(stream);
        audioChunks = [];

        mediaRecorder.ondataavailable = event => {
          if (event.data.size > 0) audioChunks.push(event.data);
        };

        mediaRecorder.onstop = () => {
          const audioBlob = new Blob(audioChunks, { type: 'audio/webm' });
          const audioUrl = URL.createObjectURL(audioBlob);
          audioPlayback.src = audioUrl;
          audioPlayback.style.display = 'block';

          const durationSec = Math.max(5, Math.round((Date.now() - recordingStartTime) / 1000));
          examState.autoDetectedSeconds = durationSec;
          saveState();
        };

        mediaRecorder.start();
        if (speechRecognizer) speechRecognizer.start();
        recordingStartTime = Date.now();

        isRecording = true;
        recordBtn.textContent = '⏹️ Stop Recording';
        recordBtn.classList.add('recording');
        recordStatus.textContent = 'Recording voice...';
      } catch (err) {
        alert('Microphone access denied or unavailable on this browser.');
      }
    } else {
      mediaRecorder.stop();
      if (speechRecognizer) speechRecognizer.stop();
      isRecording = false;
      recordBtn.textContent = '🔴 Start Recording';
      recordBtn.classList.remove('recording');
      recordStatus.textContent = 'Recording saved & evaluated automatically.';
    }
  });

  function renderExam() {

      box.className = 'q1-item';
      box.innerHTML = `
        <div class="letter-pair"><span>${idx + 1}. ${item.left}</span><span>${item.right}</span></div>
        <div class="options-row">
          ${item.options.map(opt => `
            <label class="option-btn ${examState.answers[item.id] === opt ? 'selected' : ''}" data-qid="${item.id}" data-val="${opt}">
              <input type="radio" name="${item.id}" value="${opt}" ${examState.answers[item.id] === opt ? 'checked' : ''}/>
              ${opt}
            </label>
          `).join('')}
        </div>
      `;
      q1Div.appendChild(box);
    });

    const q2Div = document.getElementById('q2Container');
    examData.q2.forEach(item => {

      lbl.className = 'seq-box';
      lbl.innerHTML = `
        <input type="checkbox" data-qid="${item.id}" ${checked ? 'checked' : ''} style="width:18px;height:18px;"/>
        <span>${item.num}. ${item.text}</span>
      `;
      q2Div.appendChild(lbl);
    });

    renderTCGrid('q3Container', examData.q3, 'q3');
    renderTCGrid('q4Container', examData.q4, 'q4');
    renderTCGrid('q5_1Container', examData.q5_1, 'q5_1');

    const q5_2Div = document.getElementById('q5_2Container');
    examData.q5_2.forEach(item => {
      const key = `q5_2_${item.num}`;
      const val = examState.answers[key] || '';
      const div = document.createElement('div');
      div.className = 'dict-item';
      div.innerHTML = `
        <div class="dict-header"><span>#${item.num}</span><button type="button" class="audio-btn" onclick="speak('${item.promptAudio}')">🔊 Play</button></div>
        <input type="text" data-qid="${key}" value="${val}" placeholder="Word..." autocorrect="off" autocapitalize="off" spellcheck="false" autocomplete="off" />
      `;
      q5_2Div.appendChild(div);
    });

    const q5_3Div = document.getElementById('q5_3Container');
    examData.q5_3.forEach(item => {

      const val = examState.answers[key] || '';

      div.className = 'dict-item';
      div.innerHTML = `
        <div class="dict-header"><span>${item.num}. <strong>${item.pattern}</strong></span><button type="button" class="audio-btn" onclick="speak('${item.audio}')">🔊 Play</button></div>
        <input type="text" maxlength="1" data-qid="${key}" value="${val}" placeholder="Letter" autocorrect="off" autocapitalize="off" spellcheck="false" autocomplete="off" />

      const savedIdx = examState.answers[key];
      const card = document.createElement('div');
      card.className = 'mcq-card';
      card.innerHTML = `
        <p>${item.q}</p>
        ${item.options.map((opt, oIdx) => `
          <label class="mcq-option">
            <input type="radio" name="${key}" data-qid="${key}" value="${oIdx}" ${savedIdx === oIdx ? 'checked' : ''} />
            ${opt}
          </label>
        `).join('')}
      `;
      q7Div.appendChild(card);
    });

    const q8Div = document.getElementById('q8Container');
    examData.q8.forEach((col, cIdx) => {
      const table = document.createElement('table');
      table.className = 'trans-table';
      table.innerHTML = `
        <thead><tr><th>English</th><th>Arabic</th></tr></thead>
        <tbody>
          ${col.map((row, rIdx) => {
            const key = `q8_${cIdx}_${rIdx}`;
            const val = examState.answers[key] || '';
            return `<tr><td><input type="text" data-qid="${key}" value="${val}" placeholder="English..." autocorrect="off" autocapitalize="off" spellcheck="false" autocomplete="off" /></td><td class="arabic-col">${row.ar}</td></tr>`;

      q8Div.appendChild(table);

    ['candidateName', 'schoolName', 'cityAddress', 'examDate', 'readSeconds', 'correctWords'].forEach(id => {
      if (examState.answers[id]) document.getElementById(id).value = examState.answers[id];
    });

    if (examState.started) document.getElementById('lockBadge').style.display = 'inline-block';

  function renderTCGrid(containerId, items, prefix) {
    const container = document.getElementById(containerId);
    items.forEach((label, idx) => {
      const key = `${prefix}_${idx}`;
      const stateVal = examState.answers[key];
      const tile = document.createElement('div');
      tile.className = 'phonics-tile';
      tile.innerHTML = `
        <div class="phonics-char">${label}</div>
        <div class="tick-cross-group">
          <button type="button" class="tc-btn ${stateVal === 'tick' ? 'active-tick' : ''}" data-qid="${key}" data-tc="tick">✓</button>
          <button type="button" class="tc-btn ${stateVal === 'cross' ? 'active-cross' : ''}" data-qid="${key}" data-tc="cross">✗</button>
        </div>
      `;
      container.appendChild(tile);
    });
  }

  function speak(word) {
    lockExamOnFirstInteraction();
    if ('speechSynthesis' in window) {
      window.speechSynthesis.cancel();
      const utter = new SpeechSynthesisUtterance(word);
      utter.lang = 'en-US';
      utter.rate = 0.8;
      window.speechSynthesis.speak(utter);
    }
  }

  // ==========================================
  // EVENT LISTENERS & AUTO-SAVE
  // ==========================================
  document.addEventListener('click', function (e) {
    if (examState.submitted) return;
    const optBtn = e.target.closest('.option-btn');
    if (optBtn) {
      lockExamOnFirstInteraction();
      const qid = optBtn.dataset.qid;
      const val = Number(optBtn.dataset.val);
      examState.answers[qid] = val;
      document.querySelectorAll(`.option-btn[data-qid="${qid}"]`).forEach(b => b.classList.remove('selected'));
      optBtn.classList.add('selected');
      saveState();
    }

    const tcBtn = e.target.closest('.tc-btn');
    if (tcBtn) {
      lockExamOnFirstInteraction();
      const qid = tcBtn.dataset.qid;
      const tc = tcBtn.dataset.tc;
      examState.answers[qid] = tc;
      const siblings = tcBtn.parentElement.querySelectorAll('.tc-btn');
      siblings[0].classList.toggle('active-tick', tc === 'tick');
      siblings[1].classList.toggle('active-cross', tc === 'cross');
      saveState();
    }
  });

  document.addEventListener('change', function (e) {
    if (examState.submitted) return;
    const target = e.target;
    if (target.type === 'checkbox' && target.dataset.qid) {
      lockExamOnFirstInteraction();
      examState.answers[target.dataset.qid] = target.checked;
      saveState();
    }
    if (target.type === 'radio' && target.dataset.qid) {
      lockExamOnFirstInteraction();
      examState.answers[target.dataset.qid] = Number(target.value);
      saveState();
    }
  });

  document.addEventListener('input', function (e) {
    if (examState.submitted) return;
    const target = e.target;
    const key = target.dataset.qid || target.id;
    if (key) {
      if (target.dataset.qid || target.id === 'readSeconds' || target.id === 'correctWords') {
        lockExamOnFirstInteraction();
      }
      examState.answers[key] = target.value;
      saveState();
    }
  });

  // Block paste in translation/dictation fields to discourage external translators
  document.addEventListener('paste', function (e) {
    if (e.target.tagName === 'INPUT' && e.target.dataset.qid) {
      e.preventDefault();
    }
  });

  // Main Countdown Timer
  setInterval(() => {
    if (examState.submitted) return;
    if (examState.started && examState.remainingSeconds > 0) {
      examState.remainingSeconds--;
      saveState();
    }
    const mins = Math.floor(examState.remainingSeconds / 60);
    const secs = examState.remainingSeconds % 60;
    document.getElementById('timerDisplay').textContent = `${String(mins).padStart(2, '0')}:${String(secs).padStart(2, '0')}`;
    if (examState.remainingSeconds === 0 && !examState.submitted) finalizeAndShowScore();
  }, 1000);

  // ==========================================
  // FINAL SCORING ENGINE (SHOWN ONLY AT END)
  // ==========================================
  function finalizeAndShowScore() {
    examState.submitted = true;
    saveState();

    document.querySelectorAll('input, button').forEach(el => el.disabled = true);

    let part1Score = 0;
    examData.q1.forEach(item => { if (examState.answers[item.id] === item.answer) part1Score += 1; });
    examData.q2.forEach(item => { if (Boolean(examState.answers[item.id]) === item.valid) part1Score += 1; });

    let q3Score = 0;
    examData.q3.forEach((_, idx) => { if (examState.answers[`q3_${idx}`] === 'tick') q3Score += 1; });

    let q4Score = 0;
    examData.q4.forEach((_, idx) => { if (examState.answers[`q4_${idx}`] === 'tick') q4Score += 1; });

    let q5Score = 0;
    examData.q5_1.forEach((_, idx) => { if (examState.answers[`q5_1_${idx}`] === 'tick') q5Score += 1; });
    examData.q5_2.forEach(item => {
      if ((examState.answers[`q5_2_${item.num}`] || '').trim().toLowerCase() === item.answer) q5Score += 1;
    });
    examData.q5_3.forEach(item => {
      if ((examState.answers[`q5_3_${item.num}`] || '').trim().toLowerCase() === item.answer) q5Score += 1;
    });

    // Q6: Use examiner manual override if entered, otherwise use automatic voice recording analysis
    let q6Score = 0;
    const manualSec = Number(examState.answers['readSeconds'] || 0);
    const manualWords = Number(examState.answers['correctWords'] || 0);
    const finalWords = manualWords > 0 ? Math.min(145, manualWords) : examState.autoDetectedWords;
    const finalSec = manualSec > 0 ? manualSec : examState.autoDetectedSeconds;

    if (finalWords > 0) {
      const pronScore = finalWords * 0.25;
      const speedScore = finalSec > 0 ? Math.min(8.75, (75 / finalSec) * 8.75) : 0;
      q6Score = Math.round((pronScore + speedScore) * 100) / 100;
    }

    let q7Score = 0;
    examData.q7.forEach((item, idx) => { if (examState.answers[`q7_${idx}`] === item.answer) q7Score += 1; });

    let rawTransCorrect = 0, totalTransItems = 0;
    examData.q8.forEach((col, cIdx) => {
      col.forEach((row, rIdx) => {
        totalTransItems++;
        if (row.en.includes((examState.answers[`q8_${cIdx}_${rIdx}`] || '').trim().toLowerCase())) rawTransCorrect++;
      });
    });
    const q8Score = Math.round(((rawTransCorrect / totalTransItems) * 33) * 100) / 100;

    const totalScore = Math.round((part1Score + q3Score + q4Score + q5Score + q6Score + q7Score + q8Score) * 100) / 100;

    document.getElementById('resultCandidateInfo').textContent = `Participant: ${examState.answers['candidateName'] || 'Candidate'}`;
    document.getElementById('finalScoreDisplay').textContent = `${totalScore} / 171`;

    document.getElementById('scoreBreakdown').innerHTML = `
      <div><span>Part 1: Alphabet</span><strong>${part1Score} / 8</strong></div>
      <div><span>Part 2 (Q3): Letter Sounds</span><strong>${q3Score} / 26</strong></div>
      <div><span>Part 2 (Q4): Syllables</span><strong>${q4Score} / 26</strong></div>
      <div><span>Part 2 (Q5): Listening & Writing</span><strong>${q5Score} / 26</strong></div>
      <div><span>Part 3 (Q6): Voice Fluency Evaluation</span><strong>${q6Score} / 45</strong></div>
      <div><span>Part 3 (Q7): Comprehension</span><strong>${q7Score} / 8</strong></div>
      <div><span>Part 4 (Q8): Translation (${rawTransCorrect}/${totalTransItems})</span><strong>${q8Score} / 33</strong></div>
    `;

    if (examState.tabSwitches > 0) {
      document.getElementById('tabSwitchWarning').textContent = `Security Log: Candidate left/switched browser tab ${examState.tabSwitches} time(s) during the exam.`;
    }

    document.getElementById('resultModal').style.display = 'flex';
  }

  document.getElementById('submitExamBtn').addEventListener('click', () => {
    if (confirm('Are you sure you want to submit the exam? Scores will be finalized and you cannot edit answers.')) {
      finalizeAndShowScore();
    }
  });

  loadState();
  renderExam();
  if (examState.submitted) finalizeAndShowScore();
</script>
